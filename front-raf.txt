let _fila = null;
const _pend = {};
function load(entity, force) {
  if (!force && S.cache[entity]) return Promise.resolve(S.cache[entity]);
  if (!force && _pend[entity]) return _pend[entity];
  if (!_fila) {
    const f = _fila = { ents: new Set() };
    f.p = Promise.resolve().then(() => { _fila = null; return run('api_carregar', [...f.ents], S.creds || null); })
      .then(r => { Object.entries(r).forEach(([e, rows]) => { S.cache[e] = rows; }); return r; })
      .finally(() => f.ents.forEach(e => { if (_pend[e] && _pend[e]._f === f) delete _pend[e]; }));
  }
  const f = _fila;
  f.ents.add(entity);
  const p = f.p.then(r => { if (!r[entity]) throw new Error('Sem permissão para ver ' + ((S.schema[entity] || {}).titulo || entity) + '.'); return r[entity]; });
  p._f = f;
  return (_pend[entity] = p);
}

/* Aplica no cache local só o que o servidor gravou/apagou — a tela se atualiza
 * na hora, sem baixar todas as listas de novo. */
function aplicar(m) {
  if (!m) { S.cache = {}; return; }
  Object.entries(m.salvos || {}).forEach(([e, regs]) => {
    if (!S.cache[e]) return;
    regs.forEach(r => {
      const i = S.cache[e].findIndex(x => x.id === r.id);
      if (i >= 0) S.cache[e][i] = r; else S.cache[e].push(r);
    });
  });
  Object.entries(m.removidos || {}).forEach(([e, ids]) => {
    if (S.cache[e]) S.cache[e] = S.cache[e].filter(x => !ids.includes(x.id));
  });
}

async function fotoPerfilPayload(file) {
  if (!file || !['image/png', 'image/jpeg'].includes(file.type)) throw new Error('Escolha uma foto PNG ou JPG.');
  const blob = await new Promise((resolve, reject) => {
    const imagem = new Image();
    const url = URL.createObjectURL(file);
    imagem.onload = () => {
      const escala = Math.min(1, 512 / Math.max(imagem.naturalWidth, imagem.naturalHeight));
      const canvas = document.createElement('canvas');
      canvas.width = Math.max(1, Math.round(imagem.naturalWidth * escala));
      canvas.height = Math.max(1, Math.round(imagem.naturalHeight * escala));
      const ctx = canvas.getContext('2d');
      ctx.fillStyle = '#ffffff';
      ctx.fillRect(0, 0, canvas.width, canvas.height);
      ctx.drawImage(imagem, 0, 0, canvas.width, canvas.height);
      URL.revokeObjectURL(url);
      canvas.toBlob(b => b ? resolve(b) : reject(new Error('Não foi possível preparar a foto.')), 'image/jpeg', 0.82);
    };
    imagem.onerror = () => { URL.revokeObjectURL(url); reject(new Error('Não foi possível abrir a imagem.')); };
    imagem.src = url;
  });
  if (blob.size > 500000) throw new Error('A foto ficou acima de 500 KB. Escolha uma imagem menor.');
  const base64 = await new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(String(reader.result).split(',')[1]);
    reader.onerror = () => reject(new Error('Não foi possível ler a foto.'));
    reader.readAsDataURL(blob);
  });
  return { mime: 'image/jpeg', base64 };
}

const LOJA_CHAVE = 'ger_login_v1';

function salvaLembrete(ui) {
  if (ui.lembrar) {
    localStorage.setItem(LOJA_CHAVE, JSON.stringify({ usuario: ui.usuario, senha: ui.senha || '', lembrar: true }));

function lembreteSalvo() {
  try {

    return salvo && typeof salvo === 'object' ? salvo : null;

}

async function renderEntity(host, entity, page) {
  const def = S.schema[entity];
  const cols = def.cols.filter(c => c.list && !c.hidden);
  const refs = [...new Set(cols.filter(c => c.t === 'ref').map(c => c.ref))];
  // lista e referências numa única ida ao servidor
  const [lista] = await Promise.all([load(entity), ...refs.map(e => load(e).catch(() => []))]);
  const rows = lista.slice().reverse();
  const filtros = cols.filter(c => (c.t === 'select' && c.opts) || c.t === 'ref');
  const podeCriar = !def.readonly && can(def.permAs || entity, 'c');
  const tabs = SUBTABS[page];
  const tamanhos = [10, 25, 50, 100];
  let pg = 1, tam = 25;

  const filtro = f => f.t === 'ref'
    ? `<div class="combo" data-ref="${esc(f.ref)}">
        <input class="combo-in" type="text" placeholder="${esc(f.l)}…" autocomplete="off" aria-label="${esc(f.l)}">
        <input type="hidden" data-f="${f.k}" data-hid>
        <svg class="chev" viewBox="0 0 24 24" aria-hidden="true"><path d="M6 9l6 6 6-6"/></svg>
        <div class="combo-sc" hidden></div>
      </div>`
    : `<select data-f="${f.k}" aria-label="${esc(f.l)}"><option value="">${esc(f.l)}: todos</option>${f.opts.map(o => `<option value="${esc(o)}">${esc(o)}</option>`).join('')}</select>`;

  host.innerHTML = `
    <div class="ph">
      <div><h1>${esc(def.titulo)}</h1><p class="sub" id="cnt"></p></div>
      <div class="tools">
        <span class="srch"><svg viewBox="0 0 24 24" aria-hidden="true"><circle cx="11" cy="11" r="6.5"/><path d="M16 16l4.5 4.5"/></svg>
          <input id="q" type="search" placeholder="Buscar em ${esc(def.titulo.toLowerCase())}" aria-label="Buscar"></span>
        ${filtros.map(filtro).join('')}
        ${podeCriar ? `<button class="btn primary" id="novo">Novo ${esc(def.singular.toLowerCase())}</button>` : ''}
      </div>
    </div>
    ${tabs ? `<div class="tabs" role="tablist">${tabs.map(t => `<button role="tab" data-t="${t}" aria-selected="${t === entity}">${esc(S.schema[t].titulo)}</button>`).join('')}</div>` : ''}
    <div class="card tw"><table class="tbl"><thead><tr>${cols.map(c => `<th class="${['money','number'].includes(c.t) ? 'r' : ''}">${esc(c.l)}</th>`).join('')}<th></th></tr></thead><tbody id="tb"></tbody></table>
      <div class="pag">
        <span id="pgI"></span>
        <span class="pgc"><button class="btn mini" id="pgP">Anterior</button><button class="btn mini" id="pgN">Próxima</button>
          <select id="pgT" aria-label="Registros por página">${tamanhos.map(n => `<option ${n === tam ? 'selected' : ''}>${n}</option>`).join('')}</select><span>por página</span></span>
      </div></div>`;

  const draw = () => {
    const q = $('#q').value.trim().toLowerCase();
    const fv = {}; host.querySelectorAll('[data-f]').forEach(s => fv[s.dataset.f] = s.value);
    const vis = rows.filter(r => (!q || Object.values(r).join(' ').toLowerCase().includes(q)) && Object.keys(fv).every(k => !fv[k] || r[k] === fv[k]));
    const max = Math.max(1, Math.ceil(vis.length / tam));
    if (pg > max) pg = max;
    const sl = vis.slice((pg - 1) * tam, pg * tam);
    $('#cnt').textContent = vis.length === rows.length
      ? `${rows.length} registro${rows.length === 1 ? '' : 's'}`
      : `${vis.length} de ${rows.length} registros`;
    $('#pgI').innerHTML = vis.length
      ? `<b>${(pg - 1) * tam + 1}–${Math.min(pg * tam, vis.length)}</b> de <b>${vis.length}</b>` + (vis.length !== rows.length ? ` (filtrados de ${rows.length})` : '')
      : '<b>0</b> registros';
    $('#pgP').disabled = pg <= 1; $('#pgN').disabled = pg >= max;
    $('#tb').innerHTML = sl.length ? sl.map(r => {
      const acoes = acoesDe(entity, r);
      return `<tr data-id="${esc(r.id)}">${cols.map(c => `<td class="${['money','number'].includes(c.t) ? 'r money' : ''} ${c.t === 'longtext' || c.k === 'titulo' || c.k === 'descricao' ? 'cut' : ''} ${c.k === 'id' ? 'id' : ''}">${cell(c, r[c.k])}</td>`).join('')}

          ? `<a class="btn ghost sm" href="${esc(a.href)}" target="_blank" rel="noopener">${esc(a.t)}</a>`
          : `<button class="btn ghost sm" data-a="${j}">${esc(a.t)}</button>`).join('')}</div></td></tr>`;
    }).join('') : `<tr><td colspan="${cols.length + 1}"><div class="empty">${rows.length
        ? '<b>Nenhum registro com esses filtros</b>Ajuste a busca ou limpe os filtros.'
        : `<b>Nada por aqui ainda</b>${podeCriar ? 'Use “Novo ' + esc(def.singular.toLowerCase()) + '” para começar.' : 'Os registros aparecem aqui assim que forem criados.'}`}</div></td></tr>`;
    $('#tb').onclick = e => {
      const tr = e.target.closest('tr[data-id]'); if (!tr) return;
      const r = rows.find(x => x.id === tr.dataset.id);
      const btn = e.target.closest('[data-a]');
      if (btn) { e.stopPropagation(); acoesDe(entity, r)[+btn.dataset.a].f(); }
      else if (!e.target.closest('a') && !def.readonly) abrirForm(entity, r);
    };
  };
  $('#q').oninput = () => { pg = 1; draw(); };
  host.querySelectorAll('[data-f]').forEach(s => s.onchange = () => { pg = 1; draw(); });
  $('#pgP').onclick = () => { if (pg > 1) { pg--; draw(); } };
  $('#pgN').onclick = () => { pg++; draw(); };
  $('#pgT').onchange = e => { tam = +e.target.value; pg = 1; draw(); };
  if ($('#novo')) $('#novo').onclick = () => abrirForm(entity, null);
  host.querySelectorAll('[data-t]').forEach(b => b.onclick = () => { S.sub[page] = b.dataset.t; go(page); });
  comboInit(host, () => { pg = 1; draw(); });
  draw();
}

function cell(c, v) {
  if (v === '' || v == null) return c.t === 'money' ? `<span style="color:var(--muted)">${money(0)}</span>` : '';
  if (c.t === 'money') return money(v);
  if (c.t === 'date') return fdate(v);
  if (c.t === 'datetime') return fdt(v);
  if (c.t === 'ref') return refCell(c.ref, v);
  if (c.t === 'url') return `<a href="${esc(v)}" target="_blank" rel="noopener" onclick="event.stopPropagation()">Abrir PDF</a>`;
  if (c.t === 'select' && ['status','tipo','prioridade','ativo'].includes(c.k)) return badge(v);
  return esc(v);
}

/* ---------- ações por linha ---------- */
function acoesDe(entity, r) {
  const A = [];
  if (entity === 'pedidos' && r.status !== 'Cancelado' && can('orcamentos', 'c') && can('pedidos', 'u')) A.push({ t: 'Gerar orçamento', f: () => converter('pedidos', r.id) });
  if (entity === 'orcamentos' && !['Recusado', 'Expirado'].includes(r.status) && can('os', 'c') && can('orcamentos', 'u')) A.push({ t: 'Gerar OS', f: () => converter('orcamentos', r.id) });
  if (entity === 'os' && r.status !== 'Cancelada' && can('financeiro', 'c') && can('os', 'u')) A.push({ t: 'Lançar entrada', f: () => converter('os', r.id) });
  if (entity === 'financeiro' && r.status === 'Pendente' && can('financeiro', 'u')) A.push({ t: 'Marcar como pago', f: () => marcarPago(r) });
  if (entity === 'materiais' && can('materiais', 'u')) A.push({ t: 'Movimentar', f: () => abrirMov(r) });
  if (['pedidos', 'orcamentos', 'os', 'financeiro'].includes(entity) && can(entity, 'r')) A.push({ t: 'Documento', f: () => abrirDoc(entity, r.id) });
  return A;
}

async function abrirForm(entity, rec) {
  const def = S.schema[entity];
  const novo = !rec || !rec.id;
  const perm = S.perms[def.permAs || entity] || '';
  const podeSalvar = novo ? perm.includes('c') : perm.includes('u');
  const dis = podeSalvar ? '' : 'disabled';

  const precisa = new Set();
  def.cols.forEach(c => { if (c.t === 'ref') precisa.add(c.ref); if (c.t === 'items') { precisa.add('materiais'); precisa.add('atividades'); precisa.add('os'); } });
  const aCarregar = [...precisa].filter(e => can((S.schema[e] || {}).permAs || e, 'r') && !(S.cache[e] || []).length);
  if (aCarregar.length) {
    busy(true, 'Carregando dados de ' + esc(def.singular.toLowerCase()) + '…');
    try { await Promise.all(aCarregar.map(e => load(e).catch(() => []))); }
    finally { busy(false); }
  }

  const data = Object.assign({}, rec || {});
  if (novo) Object.entries(def.defaults).forEach(([k, v]) => { if (data[k] == null || data[k] === '') data[k] = v === '@hoje' ? S.hoje : v; });

  const campos = def.cols.filter(c => !c.hidden && c.t !== 'id' && !(c.ro && /^(criado|atualizado)/.test(c.k)));
  let editor = null;

  const campo = c => {
    const sec = (SECOES[entity] || {})[c.k];
    const pre = sec ? `<div class="sect">${esc(sec.toUpperCase())}</div>` : '';
    const v = data[c.k] == null ? '' : data[c.k];
    const full = ['longtext', 'items'].includes(c.t) ? 'full' : '';
    const req = c.req ? '<span class="req" aria-hidden="true">*</span>' : '';
    if (c.t === 'items') return pre + `<div class="full"><div style="font-weight:600;font-size:12.5px;color:var(--ink-2);margin-bottom:6px">${esc(c.l.replace(' (JSON)', ''))}</div><div id="itensHost"></div></div>`;
    if (c.t === 'checkbox') return pre + `<label class="check-field"><input type="checkbox" data-k="${c.k}" value="Sim" ${v === 'Sim' ? 'checked' : ''} ${dis}>${esc(c.l)} ${req}</label>`;
    let inp;
    if (c.ro) inp = `<input disabled data-k="${c.k}" data-ro="1" value="${c.t === 'money' ? esc(money(v)) : esc(v)}">`;
    else if (c.t === 'longtext') inp = `<textarea rows="3" data-k="${c.k}" ${dis}>${esc(v)}</textarea>`;
    else if (c.t === 'email') inp = `<input type="email" data-k="${c.k}" value="${esc(v)}" ${dis} autocomplete="email">`;
    else if (c.t === 'select') {
      const ops = c.opts.slice(); if (v && !ops.includes(v)) ops.push(v);
      inp = `<select data-k="${c.k}" ${dis}><option value=""></option>${ops.map(o => `<option ${o === v ? 'selected' : ''}>${esc(o)}</option>`).join('')}</select>`;
    } else if (c.t === 'ref' && S.cache[c.ref]) {
      const lista = S.cache[c.ref].slice().reverse();
      const sel = lista.filter(r => r.id === v).map(r => refLabel(c.ref, r))[0] || refLabel(c.ref, { id: v });
      inp = `<div class="combo" data-ref="${esc(c.ref)}">
        <input class="combo-in" type="text" value="${esc(sel)}" placeholder="Digite para filtrar…" ${dis} autocomplete="off" aria-label="${esc(c.l)}">
        <input type="hidden" data-k="${c.k}" data-hid value="${esc(v)}">
        <svg class="chev" viewBox="0 0 24 24" aria-hidden="true"><path d="M6 9l6 6 6-6"/></svg>
        <div class="combo-sc" hidden></div>
      </div>`;
    } else if (c.t === 'money' || c.t === 'number') inp = `<input type="number" step="${c.t === 'money' ? '0.01' : 'any'}" data-k="${c.k}" value="${esc(v)}" ${dis}>`;
    else if (c.t === 'date') inp = `<input type="date" data-k="${c.k}" value="${esc(String(v).slice(0, 10))}" ${dis}>`;
    else if (c.t === 'password') inp = `<input type="password" data-k="${c.k}" value="${esc(v)}" ${dis} autocomplete="new-password">`;
    else inp = `<input type="${c.t === 'url' ? 'url' : 'text'}" data-k="${c.k}" value="${esc(v)}" ${dis}>`;
    return pre + `<label class="f ${full}">${esc(c.l)} ${req}${inp}</label>`;
  };

  const d = $('#dlgForm');
  d.innerHTML = `
    <div class="dh">
      <div><h2>${novo ? 'Novo ' + esc(def.singular.toLowerCase()) : esc(def.singular) + ' ' + esc(data.id)}</h2>
        <p class="sub">${novo ? 'Campos com * são obrigatórios.' : 'Todos os campos continuam editáveis.'}</p></div>
      <div style="display:flex;gap:8px;align-items:center">${!novo && data.status ? badge(data.status) : ''}<button class="btn ghost sm" data-x aria-label="Fechar">Fechar</button></div>
    </div>
    <div class="db"><div id="err"></div><div class="fg">${campos.map(campo).join('')}</div>
      ${!novo && data.atualizadoEm ? `<div class="foot">Criado por ${esc(data.criadoPor || '—')} em ${fdt(data.criadoEm)} · última edição por ${esc(data.atualizadoPor || '—')} em ${fdt(data.atualizadoEm)}</div>` : ''}</div>
    <div class="df">
      ${!novo && perm.includes('d') ? '<button class="btn danger grow" data-del>Excluir</button>' : ''}
      <button class="btn" data-x>${podeSalvar ? 'Cancelar' : 'Fechar'}</button>
      ${podeSalvar ? '<button class="btn primary" data-save>Salvar</button>' : ''}
    </div>`;
  comboInit(d);
  const ih = $('#itensHost', d);
  if (ih) {
    const totais = () => {
      const sub = editor.total();
      const desc = parseFloat(($('[data-k=desconto]', d) || {}).value) || 0;
      const s = $('[data-k=subtotal]', d), t = $('[data-k=total]', d);
      if (s) s.value = money(sub);
      if (t) t.value = money(Math.max(0, sub - desc));
    };
    editor = editorItens(data.itens, !podeSalvar, totais, { entity, id: novo ? '' : data.id, status: () => ($('[data-k=status]', d) || {}).value || data.status || '' });
    ih.appendChild(editor.el);
    const desc = $('[data-k=desconto]', d); if (desc) desc.oninput = totais;
    totais();
  }

  d.onchange = e => {
    const k = e.target.dataset.k;
    const referencia = e.target.matches('input[type="hidden"][data-hid]');
    if (!k || (e.target.tagName !== 'SELECT' && !referencia)) return;
    const g = x => ($(`[data-k=${x}]`, d) || {}).value || '';
    const s = (x, v) => { const el = $(`[data-k=${x}]`, d); if (el && !el.value && v != null && v !== '') el.value = v; };
    const achar = (ent, id) => (S.cache[ent] || []).find(r => r.id === id);
    if (k === 'status' && editor) editor.avisos();
    if (entity === 'pedidos' && k === 'cliente') { const cl = achar('clientes', g(k)); if (cl) s('contato', cl.contato || cl.celular || cl.telefone || ''); }
    if (entity === 'orcamentos' && k === 'pedidoId') { const p = achar('pedidos', g(k)); if (p) { s('cliente', p.cliente); s('titulo', String(p.descricao).slice(0, 120)); } }
    if (entity === 'os' && k === 'orcamentoId') {
      const o = achar('orcamentos', g(k));
      if (o) { s('cliente', o.cliente); s('pedidoId', o.pedidoId); s('descricao', o.titulo); if (!editor.count()) { editor.set(o.itens); s('desconto', o.desconto); } }
    }
    if (entity === 'financeiro' && k === 'osId') { const o = achar('os', g(k)); if (o) { const nc = refName('clientes', o.cliente); s('parte', nc); s('descricao', o.id + ' — ' + nc); s('valor', o.total); } }
  };

  d.onclick = async e => {
    if (e.target.closest('[data-x]')) { d.close(); return; }
    if (e.target.closest('[data-del]')) {
      if (!(await confirmar('Excluir ' + data.id + '?', 'Registros vinculados não são apagados, apenas ficam sem referência. Esta ação não pode ser desfeita.', 'Excluir', true))) return;
      try { const res = await com(() => run('api_delete', entity, data.id, S.creds || null), 'Excluindo…'); aplicar(res.mudancas); d.close(); toast('Registro excluído'); go(S.page); }
      catch (err) { $('#err', d).innerHTML = `<div class="msg">${esc(err.message)}</div>`; }
      return;
    }
    if (e.target.closest('[data-save]')) {
      const out = { id: novo ? '' : data.id };
      d.querySelectorAll('[data-k]').forEach(el => { if (!el.dataset.ro) out[el.dataset.k] = el.type === 'checkbox' ? (el.checked ? 'Sim' : 'Não') : el.value; });
      if (editor) out.itens = editor.get();
      const btn = e.target.closest('[data-save]'); btn.disabled = true; btn.textContent = 'Salvando…';
      try { const res = await com(() => run('api_save', entity, out, S.creds || null), 'Salvando ' + esc(def.singular.toLowerCase()) + '…'); aplicar(res.mudancas); d.close(); if (res.aviso) toast(res.aviso.msg, !res.aviso.ok, 9000); else toast('Salvo'); go(S.page); }
      catch (err) { $('#err', d).innerHTML = `<div class="msg">${esc(err.message)}</div>`; btn.disabled = false; btn.textContent = 'Salvar'; d.querySelector('.db').scrollTop = 0; }
    }
  };
  d.showModal();
}

/* ---------- editor de itens ---------- */
const STATUS_RESERVA = ['Aberta', 'Em andamento'], STATUS_ORC_ATIVO = ['Rascunho', 'Enviado'];
function editorItens(inicial, somenteLeitura, onChange, ctx) {
  const ler = s => { try { const a = JSON.parse(s || '[]'); return Array.isArray(a) ? a : []; } catch (e) { return []; } };
  let itens = ler(inicial);
  const mats = (S.cache.materiais || []).filter(m => m.ativo !== 'Não');
  const atvs = (S.cache.atividades || []).filter(a => a.ativo !== 'Não');
  const el = document.createElement('div'); el.className = 'itens';
  const dz = somenteLeitura ? 'disabled' : '';
  const soma = () => itens.reduce((s, i) => s + (parseFloat(i.qtd) || 0) * (parseFloat(i.valorUnit) || 0), 0);

  /* Estoque disponível = estoque atual − reservado por outras OS abertas/em andamento
   * (mesma regra do servidor). O aviso aparece na hora; o servidor bloqueia ao salvar. */
  const qtdMat = (lista, mid) => lista.filter(i => i.tipo === 'Material' && i.materialId === mid).reduce((a, i) => a + (parseFloat(i.qtd) || 0), 0);
  const reservado = mid => (S.cache.os || [])
    .filter(o => STATUS_RESERVA.includes(o.status) && !(ctx && ctx.entity === 'os' && o.id === ctx.id))
    .reduce((a, o) => a + qtdMat(ler(o.itens), mid), 0);
  const disponivel = m => Math.max(0, Math.round(((Number(m.estoqueAtual) || 0) - reservado(m.id)) * 1000) / 1000);
  const conferir = () => !!ctx && (ctx.entity === 'os' ? STATUS_RESERVA : ctx.entity === 'orcamentos' ? STATUS_ORC_ATIVO : []).includes(ctx.status());
  function avisos() {
    el.querySelectorAll('.it').forEach(row => {
      const it = itens[+row.dataset.i], w = row.querySelector('.itw'), q = row.querySelector('[data-f=qtd]');
      let msg = '';
      const m = it && it.tipo === 'Material' && it.materialId && (S.cache.materiais || []).find(x => x.id === it.materialId);
      if (m && conferir()) {
        const pedido = qtdMat(itens, m.id), disp = disponivel(m), un = m.unidade || 'un';
        if (pedido > disp) msg = `Estoque insuficiente: ${pedido} ${un} solicitado(s), disponível ${disp} ${un}` + (disp < (Number(m.estoqueAtual) || 0) ? ` (em estoque ${m.estoqueAtual}, o resto está reservado em outras OS)` : '') + '.';
      }
      if (w) { w.textContent = msg; w.hidden = !msg; }
      if (q) q.classList.toggle('est-bad', !!msg);
    });
  }

  function draw() {
    el.innerHTML = (itens.length ? `<div class="ith"><span>Tipo</span><span>Material / Atividade</span><span>Descrição</span><span>Qtd</span><span>Valor unit.</span><span style="text-align:right">Subtotal</span><span></span></div>` : '') +
      itens.map((it, i) => `<div class="it" data-i="${i}">
        <label class="itf"><span>Tipo</span><select data-f="tipo" ${dz} aria-label="Tipo"><option ${it.tipo !== 'Material' ? 'selected' : ''}>Serviço</option><option ${it.tipo === 'Material' ? 'selected' : ''}>Material</option></select></label>
        <label class="itf"><span>${it.tipo === 'Material' ? 'Material' : 'Atividade'}</span>${it.tipo === 'Material'
          ? `<select data-f="materialId" ${dz} aria-label="Material"><option value="">Escolher…</option>${mats.map(m => `<option value="${esc(m.id)}" ${m.id === it.materialId ? 'selected' : ''}>${esc((m.sku ? m.sku + ' · ' : '') + m.nome + ' — disp. ' + disponivel(m) + ' ' + (m.unidade || 'un'))}</option>`).join('')}</select>`
          : `<select data-f="atividadeId" ${dz} aria-label="Atividade"><option value="">Escolher…</option>${atvs.map(a => `<option value="${esc(a.id)}" ${a.id === it.atividadeId ? 'selected' : ''}>${esc(a.codigo + ' · ' + a.descricao)}</option>`).join('')}</select>`}</label>
        <label class="itf d"><span>Descrição</span><input data-f="descricao" value="${esc(it.descricao || '')}" placeholder="Descrição do item" ${dz} aria-label="Descrição"></label>
        <label class="itf"><span>Quantidade</span><input data-f="qtd" type="number" step="any" min="0" value="${esc(it.qtd == null ? '' : it.qtd)}" ${dz} aria-label="Quantidade"></label>
        <label class="itf"><span>Valor unitário</span><input data-f="valorUnit" type="number" step="0.01" min="0" value="${esc(it.valorUnit == null ? '' : it.valorUnit)}" ${dz} aria-label="Valor unitário"></label>
        <span class="rs">${money((parseFloat(it.qtd) || 0) * (parseFloat(it.valorUnit) || 0))}</span>
        ${somenteLeitura ? '<span></span>' : '<button type="button" class="x" data-rm aria-label="Remover item">×</button>'}
        <div class="itw" hidden></div>
      </div>`).join('') +
      (itens.length ? '' : `<p style="margin:4px 0 10px;color:var(--muted);font-size:13px">Nenhum item lançado${somenteLeitura ? '.' : '. Adicione serviços ou materiais para compor o valor.'}</p>`) +
      (somenteLeitura ? '' : '<button type="button" class="btn sm" data-add>Adicionar item</button>') +
      `<div class="itsum"><span>Soma dos itens</span><b id="itSub">${money(soma())}</b></div>`;
    avisos();
  }

  el.addEventListener('input', e => {
    const row = e.target.closest('.it'); if (!row || !e.target.dataset.f) return;
    const it = itens[+row.dataset.i];
    it[e.target.dataset.f] = e.target.value;
    row.querySelector('.rs').textContent = money((parseFloat(it.qtd) || 0) * (parseFloat(it.valorUnit) || 0));
    $('#itSub', el).textContent = money(soma()); onChange();
    if (e.target.dataset.f === 'qtd') avisos();
  });
  el.addEventListener('change', e => {
    const row = e.target.closest('.it'); if (!row) return;
    const it = itens[+row.dataset.i], f = e.target.dataset.f;
    if (f === 'tipo') {
      it.tipo = e.target.value;
      const nomeCampo = it.tipo === 'Material' ? 'atividadeId' : 'materialId';
      it[nomeCampo] = '';
      draw(); onChange();
    }
    if (f === 'materialId') {
      it.materialId = e.target.value;
      const m = mats.find(x => x.id === it.materialId);
      if (m) { it.descricao = m.nome; if (!parseFloat(it.valorUnit)) it.valorUnit = m.preco || 0; if (!it.qtd) it.qtd = 1; }
      draw(); onChange();
    }
    if (f === 'atividadeId') {
      it.atividadeId = e.target.value;
      const a = atvs.find(x => x.id === it.atividadeId);
      if (a) { it.descricao = a.descricao; if (!parseFloat(it.valorUnit)) it.valorUnit = Number(a.valor) || 0; if (!it.qtd) it.qtd = 1; }
      draw(); onChange();
    }
  });
  el.addEventListener('click', e => {
    if (e.target.closest('[data-add]')) { itens.push({ tipo: 'Serviço', materialId: '', atividadeId: '', descricao: '', qtd: 1, valorUnit: 0 }); draw(); onChange(); el.querySelector('.it:last-of-type [data-f=descricao]').focus(); }
    const rm = e.target.closest('[data-rm]'); if (rm) { itens.splice(+rm.closest('.it').dataset.i, 1); draw(); onChange(); }
  });
  draw();
  return {
    el, total: soma, count: () => itens.length, avisos,
    set: s => { itens = ler(s); draw(); onChange(); },
    get: () => JSON.stringify(itens.filter(i => i.descricao || i.materialId || i.atividadeId).map(i => ({
      tipo: i.tipo === 'Material' ? 'Material' : 'Serviço', materialId: i.materialId || '', atividadeId: i.atividadeId || '', descricao: i.descricao || '',
      qtd: parseFloat(i.qtd) || 0, valorUnit: parseFloat(i.valorUnit) || 0 })))
  };
}

async function abrirDoc(entity, id) {
  const d = $('#dlgDoc');
  d.innerHTML = `<div class="dh"><div><h2 id="dt">Preparando documento…</h2><p class="sub" id="ds">Pré-visualização do modelo que vai para o PDF.</p></div>
      <button class="btn ghost sm" data-x>Fechar</button></div>
    <div class="db"><div id="err"></div><iframe class="doc" id="ifr" title="Pré-visualização do documento"></iframe><div id="dres" class="foot"></div></div>
    <div class="df"><button class="btn" data-print disabled>Imprimir</button><button class="btn primary" data-pdf disabled>Salvar PDF no Drive</button></div>`;
  d.showModal();
  d.onclick = async e => {
    if (e.target.closest('[data-x]')) return d.close();
    if (e.target.closest('[data-print]')) { imprimirDoc($('#ifr', d)); return; }
    const b = e.target.closest('[data-pdf]'); if (!b) return;
    b.disabled = true; b.textContent = 'Gerando PDF…';
    try {
      const r = await com(() => run('api_gerarDocumento', entity, id, S.creds || null), 'Gerando PDF…');
      delete S.cache[entity];
      $('#dres', d).innerHTML = `PDF salvo no Drive. <a href="${esc(r.url)}" target="_blank" rel="noopener">Abrir arquivo</a>`;
      toast('PDF gerado');
    } catch (err) { $('#err', d).innerHTML = `<div class="msg">${esc(err.message)}</div>`; }
    b.disabled = false; b.textContent = 'Salvar PDF no Drive';
  };
  try {
    const r = await com(() => run('api_previewDocumento', entity, id, S.creds || null), 'Gerando documento…');
    $('#dt', d).textContent = r.titulo;
    $('#ifr', d).srcdoc = r.html;
    d.querySelectorAll('[data-pdf],[data-print]').forEach(b => b.disabled = false);
  } catch (err) { $('#dt', d).textContent = 'Documento'; $('#err', d).innerHTML = `<div class="msg">${esc(err.message)}</div>`; }
}

/** Imprime a pré-visualização. Dentro de um iframe incorporado alguns
 *  navegadores bloqueiam o print, então caímos para uma janela separada. */
function imprimirDoc(ifr) {
  try {
    const w = ifr.contentWindow;
    if (!w) throw new Error('sem acesso');
    w.focus(); w.print();
  } catch (e) {
    const w = window.open('', '_blank');
    if (!w) return toast('Libere as janelas pop-up para imprimir, ou salve o PDF no Drive.', true);
    w.document.write(ifr.getAttribute('srcdoc') || '');
    w.document.close(); w.focus();
    setTimeout(() => w.print(), 400);
  }
}

            ${logo.temLogo ? '<button class="btn danger" id="rm">Remover</button>' : ''}
          </div>
        </div>
        <p class="hint" style="margin-top:12px">A imagem é gravada no Drive, aparece no topo do sistema e é embutida em cada PDF gerado.</p>
      </section>
    </div>`;

  const err = m => $('#err', host).innerHTML = m ? `<div class="msg">${esc(m)}</div>` : '';

  $('#cfgSave').onclick = async () => {
    const o = {}; host.querySelectorAll('[data-c]').forEach(i => o[i.dataset.c] = i.value);
    try {
      S.config = await com(() => run('api_saveConfig', o, S.creds || null), 'Salvando configurações…');
      renderShell();
      err(''); toast('Configurações salvas');
    } catch (e) { err(e.message); }
  };

  $('#mailTest').onclick = async () => {
    const box = $('#mailRes', host);
    const linha = (ok, titulo, texto) => `<div style="display:flex;gap:10px;padding:9px 11px;border-radius:var(--r-md);margin-bottom:6px;border:1px solid ${ok ? '#CFE6DA' : '#F2CFCB'};background:${ok ? '#EAF5EF' : '#FBE7E4'}">
      <strong style="color:${ok ? 'var(--ok)' : 'var(--bad)'};flex:none">${ok ? '✓' : '✗'}</strong>
      <div><div style="font-weight:600">${esc(titulo)}</div><div style="font-size:13px;color:var(--ink-2);word-break:break-word">${esc(texto)}</div></div></div>`;
    try {
      const r = await com(() => run('api_testarEmail', $('#mailTo', host).value.trim(), S.creds || null), 'Disparando e-mail de teste…');
      const u = r.ultimoAlerta;
      box.innerHTML =
        `<p style="margin:0 0 8px;font-weight:600;color:${r.sucesso ? 'var(--ok)' : 'var(--bad)'}">${r.sucesso ? 'Teste concluído: e-mail enviado.' : 'Teste falhou — veja a etapa marcada com ✗.'}</p>` +

        `<p class="hint" style="margin-top:10px">Último alerta automático de estoque: ${u ? `${esc(u.quando)} · ${esc(u.material)} · ${u.status === 'ok' ? 'enviado' : 'FALHOU'} — ${esc(u.detalhe)}` : 'nenhum registrado ainda (nenhum material ficou abaixo do mínimo desde esta versão).'}</p>`;
      toast(r.sucesso ? 'E-mail de teste enviado' : 'Falha no teste de e-mail', !r.sucesso);
    } catch (e) {
      // Erro antes do diagnóstico (servidor não respondeu, função não existe na implantação, sem permissão de admin...)
      const m = e.message || String(e);
      const dica = /api_testarEmail|not a function|is not defined|não é uma função|script function not found/i.test(m)
        ? 'A implantação publicada ainda não tem a função api_testarEmail. Salve o script e faça Implantar > Gerenciar implantações > editar > Versão "Nova versão".'
        : /admin|permiss|acesso/i.test(m) ? 'Somente administradores podem disparar o teste.'
        : 'O servidor não chegou a executar o diagnóstico.';
      box.innerHTML = linha(false, 'Não foi possível executar o teste', `${dica} [Erro: ${m}]`);
      toast('Falha no teste de e-mail', true);
    }
  };

  async function subirLogotipo(file, base64) {
    const creds = S.creds || null;
    const mime = file.type || (/(\.png)$/i.test(file.name) ? 'image/png' : /(\.gif)$/i.test(file.name) ? 'image/gif' : 'image/jpeg');
    if (!['image/png', 'image/jpeg', 'image/gif'].includes(mime)) throw new Error('Use uma imagem PNG, JPG ou GIF.');
    if (base64.length <= 180000) {
      return await run('api_uploadLogo', { nome: file.name, mime, base64 }, creds);
    }
    const tamanhoParte = 180000;
    const totalPartes = Math.ceil(base64.length / tamanhoParte);
    const { uploadId } = await run('api_logoIniciar', file.name, mime, file.size, totalPartes, creds);
    try {
      for (let indice = 0; indice < totalPartes; indice++) {
        await run('api_logoEnviar', uploadId, indice, base64.slice(indice * tamanhoParte, (indice + 1) * tamanhoParte), creds);
      }
      return await run('api_logoFinalizar', uploadId, creds);
    } catch (e) {
      try { await run('api_logoCancelar', uploadId, creds); } catch (ignorado) {}
      throw e;
    }
  }

  const enviar = async file => {
    if (!file) return;
    if (file.size > 1500000) return err('Imagem muito grande. Envie um arquivo de até 1,5 MB.');
    const btn = $('#pick', host); const txt = btn.textContent; btn.disabled = true; btn.textContent = 'Enviando…';
    try {
      const base64 = await new Promise((res, rej) => {
        const r = new FileReader();
        r.onload = () => res(String(r.result).split(',')[1]);
        r.onerror = () => rej(new Error('Não foi possível ler o arquivo.'));
        r.readAsDataURL(file);
      });
      const resultado = await com(() => subirLogotipo(file, base64), 'Enviando logotipo…');
      S.logo = resultado.dataUri || '';
      S.temLogo = !!resultado.temLogo;
      renderShell();
      err(''); toast('Logotipo atualizado'); go('config');
    } catch (e) { err(e.message); btn.disabled = false; btn.textContent = txt; }
  };

  $('#pick').onclick = () => $('#file', host).click();
  $('#file').onchange = e => enviar(e.target.files[0]);
  if ($('#rm', host)) $('#rm').onclick = async () => {
    if (!(await confirmar('Remover logotipo?', 'Os próximos documentos passam a usar o nome da empresa no cabeçalho.', 'Remover', true))) return;
    try {
      const resultado = await com(() => run('api_removerLogo', S.creds || null), 'Removendo logotipo…');
      S.logo = resultado.dataUri || '';
      S.temLogo = !!resultado.temLogo;
      renderShell(); toast('Logotipo removido'); go('config');
    } catch (e) { err(e.message); }
  };

  const drop = $('#drop', host);
  ['dragenter', 'dragover'].forEach(ev => drop.addEventListener(ev, e => { e.preventDefault(); drop.classList.add('on'); }));
  ['dragleave', 'drop'].forEach(ev => drop.addEventListener(ev, e => { e.preventDefault(); drop.classList.remove('on'); }));
  drop.addEventListener('drop', e => enviar(e.dataTransfer.files[0]));
}

/* Filtros do dashboard: começam no mês vigente a cada login e ficam
 * guardados enquanto a sessão estiver aberta. */
const isoLocal = d => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
const PERIODOS = {
  mes:      ['Este mês',        h => [new Date(h.getFullYear(), h.getMonth(), 1), new Date(h.getFullYear(), h.getMonth() + 1, 0)]],
  mesAnt:   ['Mês passado',     h => [new Date(h.getFullYear(), h.getMonth() - 1, 1), new Date(h.getFullYear(), h.getMonth(), 0)]],
  d7:       ['Últimos 7 dias',  h => [new Date(h.getFullYear(), h.getMonth(), h.getDate() - 6), h]],
  d30:      ['Últimos 30 dias', h => [new Date(h.getFullYear(), h.getMonth(), h.getDate() - 29), h]],
  tri:      ['Este trimestre',  h => [new Date(h.getFullYear(), Math.floor(h.getMonth() / 3) * 3, 1), new Date(h.getFullYear(), Math.floor(h.getMonth() / 3) * 3 + 3, 0)]],
  ano:      ['Este ano',        h => [new Date(h.getFullYear(), 0, 1), new Date(h.getFullYear(), 11, 31)]],
  anoAnt:   ['Ano passado',     h => [new Date(h.getFullYear() - 1, 0, 1), new Date(h.getFullYear() - 1, 11, 31)]],
  personal: ['Personalizado',   null]
};
function filtroPadrao() {
  const [de, ate] = PERIODOS.mes[1](new Date());
  return { periodo: 'mes', de: isoLocal(de), ate: isoLocal(ate), cliente: '', parceiro: '' };
}

async function renderDash(host) {
  if (!S.dash) S.dash = filtroPadrao();
  const F0 = S.dash;
  const temParc = can('parceiros', 'r');
  await Promise.all(['clientes'].concat(temParc ? ['parceiros'] : []).map(e => load(e).catch(() => [])));
  const combo = (ref, k, ph) => {
    const atual = (S.cache[ref] || []).find(r => r.id === F0[k]);
    return `<div class="combo" data-ref="${ref}">
      <input class="combo-in" type="text" placeholder="${ph}" autocomplete="off" aria-label="${ph}" value="${atual ? esc(refLabel(ref, atual)) : ''}">
      <input type="hidden" data-f="${k}" data-hid value="${esc(F0[k])}">
      <svg class="chev" viewBox="0 0 24 24" aria-hidden="true"><path d="M6 9l6 6 6-6"/></svg>
      <div class="combo-sc" hidden></div></div>`;
  };

  host.innerHTML = `
    <div class="ph"><div><h1>Indicadores</h1><p class="sub" id="dSub">${esc(S.config.empresa_nome || S.app)}</p></div></div>
    <div class="dfil card">
      <label>Período<select id="fPer">${Object.entries(PERIODOS).map(([k, [t]]) => `<option value="${k}" ${k === F0.periodo ? 'selected' : ''}>${t}</option>`).join('')}</select></label>
      <label>De<input type="date" id="fDe" value="${esc(F0.de)}"></label>
      <label>Até<input type="date" id="fAte" value="${esc(F0.ate)}"></label>
      <label class="cb">Cliente${combo('clientes', 'cliente', 'Todos os clientes')}</label>
      ${temParc ? `<label class="cb">Parceiro${combo('parceiros', 'parceiro', 'Todos os parceiros')}</label>` : ''}
      <button class="btn ghost sm" id="fLimpa" type="button">Limpar filtros</button>
    </div>
    <div id="dBody"><div class="load">Carregando…</div></div>`;

  const corpo = $('#dBody', host);
  let seq = 0;
  const atualizar = async () => {
    const meu = ++seq;
    const F = S.dash;
    if (F.de && F.ate && F.de > F.ate) { corpo.innerHTML = '<div class="msg">A data inicial é maior que a final.</div>'; return; }
    corpo.style.opacity = '.55';
    try {
      const d = await run('api_dashboard', S.creds || null, { de: F.de, ate: F.ate, cliente: F.cliente, parceiro: F.parceiro });
      if (meu !== seq) return; // chegou resposta de um filtro antigo
      desenharDash(corpo, d);
      const extra = [F.cliente && refName('clientes', F.cliente), F.parceiro && refName('parceiros', F.parceiro)].filter(Boolean).join(' · ');
      $('#dSub', host).textContent = `${S.config.empresa_nome || S.app} · ${fdate(d.filtro.de)} a ${fdate(d.filtro.ate)}${extra ? ' · ' + extra : ''}`;
    } catch (e) {
      if (meu === seq) corpo.innerHTML = `<div class="msg">${esc(e.message)}</div>`;
    } finally { if (meu === seq) corpo.style.opacity = ''; }
  };

  $('#fPer', host).onchange = e => {
    S.dash.periodo = e.target.value;
    const fn = PERIODOS[e.target.value][1];
    if (fn) {
      const [de, ate] = fn(new Date());
      S.dash.de = isoLocal(de); S.dash.ate = isoLocal(ate);
      $('#fDe', host).value = S.dash.de; $('#fAte', host).value = S.dash.ate;
    }
    atualizar();
  };
  ['fDe', 'fAte'].forEach(id => $('#' + id, host).onchange = e => {
    S.dash[id === 'fDe' ? 'de' : 'ate'] = e.target.value;
    S.dash.periodo = 'personal'; $('#fPer', host).value = 'personal';
    if (S.dash.de && S.dash.ate) atualizar();
  });
  comboInit(host, () => {
    host.querySelectorAll('.dfil [data-f]').forEach(i => { S.dash[i.dataset.f] = i.value; });
    atualizar();
  });
  $('#fLimpa', host).onclick = () => { S.dash = filtroPadrao(); renderDash(host); };
  await atualizar();
}

function desenharDash(host, d) {
  const P = d.pipeline, F = d.fin, O = d.op;
  const est = [
    { id: 'pedidos', n: P.pedidos.qtd, t: 'Pedidos aguardando orçamento', s: P.pedidos.qtd ? 'Novos ou em análise, criados no período' : 'Nenhum na fila no período' },
    { id: 'orcamentos', n: P.orcamentos.qtd, t: 'Orçamentos em aberto', s: money(P.orcamentos.valor) + ' propostos no período' },
    { id: 'os', n: P.os.qtd, t: 'Ordens de serviço em execução', s: money(P.os.valor) + ' abertas no período' }
  ];
  if (F) est.push({ id: 'financeiro', n: money(F.aReceber), money: true, t: 'A receber no período', s: F.vencido > 0 ? money(F.vencido) + ' vencido até hoje' : 'Nada vencido', bad: F.vencido > 0 });

  const max = F ? Math.max(1, ...F.serie.flatMap(x => [x.entradas, x.saidas])) : 1;
  const res = F ? F.recebidoMes - F.pagoMes : 0;
  const naFaixa = m => m >= d.filtro.de.slice(0, 7) && m <= d.filtro.ate.slice(0, 7);

  host.innerHTML = `
    <div class="rail" role="list">${est.filter(s => can(s.id, 'r')).map(s => `
      <button class="st" role="listitem" data-go="${s.id}">
        <div class="n ${s.money ? 'money' : ''}">${s.n}</div><div class="t">${esc(s.t)}</div><div class="s ${s.bad ? 'bad' : ''}">${esc(s.s)}</div>
      </button>`).join('')}</div>

    ${F ? `<div class="figs">
      <div class="fig"><div class="v pos">${money(F.recebidoMes)}</div><div class="k">Recebido no período</div></div>
      <div class="fig"><div class="v">${money(F.pagoMes)}</div><div class="k">Pago no período</div></div>

    <div class="cols">
      ${F ? `<section class="pn"><h2>Entradas e saídas por mês</h2>

          <div class="bp" title="${fmes(x.mes)} — entradas ${money(x.entradas)}, saídas ${money(x.saidas)}">
            <div class="bar in" style="height:${x.entradas / max * 100}%"></div><div class="bar out" style="height:${x.saidas / max * 100}%"></div></div>
          <div class="bl">${fmes(x.mes)}</div></div>`).join('')}</div>
        <div class="leg"><span><i style="background:var(--navy-2)"></i>Entradas pagas</span><span><i style="background:var(--brand)"></i>Saídas pagas</span><span style="opacity:.6">Meses apagados: fora do período</span></div></section>` : ''}
      <section class="pn"><h2>Operação no período</h2><dl class="m">
        <dt>Conversão de orçamentos</dt><dd>${pct(O.conversao)}</dd>
        <dt>Ticket médio por OS</dt><dd>${O.ticketMedio == null ? '—' : money(O.ticketMedio)}</dd>
        <dt>Margem sobre materiais</dt><dd>${pct(O.margem)}</dd>
        <dt>OS concluídas</dt><dd>${O.osConcluidas}</dd>
        <dt>OS atrasadas (hoje)</dt><dd style="${O.osAtrasadas ? 'color:var(--bad)' : ''}">${O.osAtrasadas}</dd>
        <dt>Valor em estoque (custo, hoje)</dt><dd>${money(O.valorEstoque)}</dd></dl></section>
    </div>

    <div class="cols" style="grid-template-columns:repeat(auto-fit,minmax(286px,1fr))">
      <section class="pn"><h2>Materiais abaixo do mínimo</h2>${d.estoqueBaixo.length ? `<ul class="ls">${d.estoqueBaixo.map(m =>
